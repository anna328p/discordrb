def extension
  if png? || apng?
    'png'
  elsif lottie?
    'json'
  elsif gif?
    'gif'
  end
end

def url(query: true)
  url = API.sticker_url(@id, extension)

  gif? && query ? ("#{url}?size=4096") : url
end
