def option(title, description: nil, channels: [], roles: [], emoji: nil)
  @options << {
    title: title,
    description: description,
    role_ids: [*roles].map(&:resolve_id),
    channel_ids: [*channels].map(&:resolve_id),
    **Option.convert_emoji(emoji, @server)
  }
end
