def ==(other)
  return false unless other.is_a?(Trigger)

  @type == other.type && @regex_patterns == other.regex_patterns && @mention_limit == other.mention_limit &&
    @mention_raid_protection == other.mention_raid_protection? && @keyword_filter == other.keyword_filter &&
    @keyword_presets == other.keyword_presets && @exempt_keywords == other.exempt_keywords
end
